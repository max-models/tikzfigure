\documentclass[border=10pt]{standalone}
\PassOptionsToPackage{dvipsnames,svgnames,x11names}{xcolor}
\usepackage{tikz}
\usepackage{pgfplots}
\pgfplotsset{compat=newest}
\usepgfplotslibrary{groupplots}
\usetikzlibrary{arrows.meta,decorations.pathmorphing,decorations.pathreplacing}
\begin{document}
\begin{tikzpicture}
    \draw[decorate, decoration={snake, amplitude=1mm}] (0, 0) -- (3, 0);
    \draw[decorate, decoration={brace, amplitude=5pt, raise=2pt}] (0, 0.5) -- (3, 0.5);
    \draw[decoration={zigzag}, decorate, red] (0, -1) -- (3, -1);
    \node at (1.5, 1.2) {brace};
\end{tikzpicture}
\end{document}
