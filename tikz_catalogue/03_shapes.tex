    \filldraw[fill=yellow!40, draw=black] (1.5, 0) rectangle (2.5, 1.5);
    \draw[blue] (0, 0) arc[start angle=0, end angle=120, radius=1.5];
    \draw (0, 0) ellipse (2cm and 1cm);
    \draw[red] (2, -0.5) circle[radius=3mm];
