\documentclass[border=10pt]{standalone}
\PassOptionsToPackage{dvipsnames,svgnames,x11names}{xcolor}
\usepackage{tikz}
\usepackage{pgfplots}
\pgfplotsset{compat=newest}
\usepgfplotslibrary{groupplots}
\usetikzlibrary{arrows.meta}
\begin{document}
\begin{tikzpicture}
    \foreach \x in {0,1,2,3} {
        \draw (\x, 0) -- (\x, 1);
    }
    \foreach \i in {1,...,5} {
        \node[circle, fill=red!\i0] at (\i, 2) {};
    }
    \foreach \x in {0,1} {
        \foreach \y in {0,1} {
            \fill (\x, {\y + 3}) circle (2pt);
        }
    }
    \foreach \x/\label in {0/a, 1/b, 2/c}
        \node at (\x, -1) {\label};
    \foreach \angle [count=\n] in {0,90,180,270} {
        \node at (\angle:1.5cm) {\n};
    }
\end{tikzpicture}
\end{document}
