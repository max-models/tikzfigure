    \node[draw] (c) at (3, -1) {C};
    \draw[->, thick] (a) -- (b);
    \draw[-{Stealth[length=3mm]}, blue] (a) |- (c);
    \draw[dashed] (b.south) -- (c.north);
    \draw[red, ->] (b) to[bend left=30] node[right] {loop} (c);
    \draw (0, -2) -- (1, -2.5) -- (2, -2) -- cycle;
    \fill[orange!40] (3, -2) -- (4, -2) -| (4, -3) -- cycle;
